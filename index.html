// 1. 单个号码直接录入云端 async function addSingleNumber() { const number = document.getElementById('singleNumberInput').value.trim(); const source = document.getElementById('singleSourceInput').value.trim() || '手动录入'; const msgBox = document.getElementById('singleNumberMsg'); if (!number) { msgBox.innerHTML = '请输入号码！'; return; } // 格式校验与清洗 const cleanNum = number.replace(/[^\d-]/g, ''); try { msgBox.innerHTML = '正在写入云端...'; // 调用你的云数据库写入接口（以你的数据库逻辑为准） await saveToDatabase([{ number: cleanNum, source: source, create_time: new Date().toISOString() }]); msgBox.innerHTML = `✔ 号码 ${cleanNum} 已成功入库！`; document.getElementById('singleNumberInput').value = ''; } catch (err) { msgBox.innerHTML = `录入失败: ${err.message}`; } } // 2. 临时存储从单文件解析出的号码列表 let parsedBatchData = []; // 3. 处理单文件上传 (.txt, .jsonl, .csv) function handleSingleFile(event) { const file = event.target.files[0]; if (!file) return; const statusMsg = document.getElementById('batchStatusMsg'); statusMsg.innerText = `正在读取文件: ${file.name}...`; const reader = new FileReader(); reader.onload = function(e) { const content = e.target.result; // 正则提取所有符合规则的号码：支持 xxx-xxxx 短号、10位、11位等纯数字 const regex = /(?:\b\d{3}-\d{4}\b|\b1\d{10}\b|\b\d{10,11}\b)/g; const matches = content.match(regex) || []; // 数组去重 const uniqueNumbers = [...new Set(matches)]; parsedBatchData = uniqueNumbers; statusMsg.innerHTML = `成功识别到 ${uniqueNumbers.length} 个有效号码，点击下方按钮即可写入云端。`; }; reader.readAsText(file); } // 4. 批量提交到云端 async function submitBatchToCloud() { const source = document.getElementById('batchSourceInput').value.trim() || '批量导入'; const statusMsg = document.getElementById('batchStatusMsg'); if (parsedBatchData.length === 0) { alert('暂无解析出的号码，请先选择文件！'); return; } statusMsg.innerHTML = `正在批量写入 ${parsedBatchData.length} 个号码到云端，请稍候...`; try { const payload = parsedBatchData.map(num => ({ number: num, source: source, create_time: new Date().toISOString() })); await saveToDatabase(payload); statusMsg.innerHTML = `🎉 写入完成！共入库 ${parsedBatchData.length} 条记录。`; parsedBatchData = []; } catch (err) { statusMsg.innerHTML = `写入出错: ${err.message}`; } }
